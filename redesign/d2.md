## TSX (tek pastable kod bloğu; render.tsx'in D2Unit + D2TrayPrints'i bunu ile degistirilir)
- Import yok. A / G / MS / INK / shade / Pr-* / FakeKnob / SegDisplay / useStore mevcut kapsamda.
- Sim etkilesimi: `sim.dev.zr.elapsed` SegDisplay'da korundu.

```tsx
function D2Unit({ sim }: { sim: SimOutput | null }) {
  const u = G.D2_UNIT;
  const sec = sim ? (sim.dev.zr.elapsed / 1000) : 0;
  return (
    <g>
      {/* cihaza ait siyah kablo stipu: govde ust-sol'dan yukari cikip kare disina (sim kablosu degil) */}
      <path d={`M ${u.cableStub.x - 10} ${u.cableStub.y - 16} Q ${u.cableStub.x - 20} ${u.cableStub.y + 12} ${u.body.x + 20} ${u.body.y + 6}`} stroke="#1a1a1c" strokeWidth={4.5} fill="none" strokeLinecap="round" />
      <ellipse cx={u.cableStub.x - 10} cy={u.cableStub.y - 18} rx={7} ry={3} fill="#23231f" />
      {/* govde: foto'da serin gri (#8A939C), beyaz degil */}
      <rect x={u.body.x} y={u.body.y} width={u.body.w} height={u.body.h} rx={4} fill="#8A939C" stroke="#3a3d42" strokeWidth={1.5} />
      {/* ust vida cubugu + 4 vita (kilitli konum) */}
      <rect x={u.body.x + 4} y={u.body.y + 3} width={u.body.w - 8} height={7} rx={2} fill="#6e7377" stroke="#45494e" strokeWidth={0.8} />
      {[0, 1, 2, 3].map((i) => (
        <circle key={`t${i}`} cx={u.body.x + 14 + i * ((u.body.w - 28) / 3)} cy={u.body.y + 6.5} r={2} fill="#4a4f52" stroke="#101014" strokeWidth={0.6} />
      ))}
      {/* alt vida cubugu + 4 vita */}
      <rect x={u.body.x + 4} y={u.body.y + u.body.h - 10} width={u.body.w - 8} height={7} rx={2} fill="#6e7377" stroke="#45494e" strokeWidth={0.8} />
      {[0, 1, 2, 3].map((i) => (
        <circle key={`b${i}`} cx={u.body.x + 14 + i * ((u.body.w - 28) / 3)} cy={u.body.y + u.body.h - 6.5} r={2} fill="#4a4f52" stroke="#101014" strokeWidth={0.6} />
      ))}
      {/* LCD: koyu bezel + acik mavi-gri ekran; Segment renkleri SegDisplay'ta (koru) */}
      <rect x={u.lcd.x} y={u.lcd.y} width={u.lcd.w} height={u.lcd.h} rx={2} fill="#1a1a1c" stroke="#0f1012" strokeWidth={1.2} />
      <rect x={u.lcd.x + 1} y={u.lcd.y + 1} width={u.lcd.w - 2} height={u.lcd.h - 2} fill="#8291A6" />
      <SegDisplay x={u.lcd.x + 2} y={u.lcd.y + 1} h={u.lcd.h - 3} value={sec.toFixed(1)} />
      {/* 4 koyu yuvarlak dugme: foto'da LCD'in daha asaginda (y≈498), render'in y441 yerine */}
      {[0, 1, 2, 3].map((i) => (
        <circle key={i} cx={u.lcd.x + 4 + i * 10} cy={u.lcd.y + u.lcd.h + 70} r={3} fill="#101014" stroke="#000" strokeWidth={0.6} />
      ))}
      {/* govde sag-ust ZR mini yazisi */}
      <text x={u.body.x + u.body.w - 8} y={u.body.y + 18} textAnchor="end" fontFamily={MS} fontSize={9} fontWeight={700} fill="#5c5f64">ZR</text>
    </g>
  );
}

function D2TrayPrints() {
  const t = G.D2_TRAY;
  return (
    <g>
      {/* ust sira: PrTimed, d2.no.a/b arasinda, merkezinin ⊤ isareti */}
      <PrTimed a={t.timed.a.x} b={t.timed.b.x} y={t.timed.a.y} />
      {/* dik bobin dikdortgeninin ICINE "ZR" (showX=false) */}
      <PrCoil a={t.coil.a.x} b={t.coil.b.x} y={(t.coil.a.y + t.coil.b.y) / 2} label="ZR" showX={false} />
      {/* iki kopru: ust sira DUZ hat, alt sira dalgali */}
      <path d={`M ${t.rowX.nc} ${t.rows[0].y} L ${t.rowX.no} ${t.rows[0].y}`} stroke={INK} strokeWidth={1.7} strokeLinecap="round" fill="none" />
      <PrWave nc={t.rowX.nc} com={t.rowX.com} no={t.rowX.no} y={t.rows[1].y} />
    </g>
  );
}
```
